```python
# OSPF Neighbor States (Critical to understand)
OSPF_STATES = [
    "Down",      # No hello packets received
    "Attempt",   # Trying to establish communication
    "Init",      # Hello received, but not bidirectional
    "2-Way",     # Bidirectional communication established
    "ExStart",   # Master/Slave relationship negotiated
    "Exchange",  # Database description packets exchanged
    "Loading",   # LSR and LSU packets exchanged
    "Full"       # Databases synchronized
]

# LSA Types on Switches
LSA_TYPES = {
    1: "Router LSA - Describes router's links",
    2: "Network LSA - Generated by DR for broadcast networks",
    3: "Summary LSA - Inter-area routes",
    4: "ASBR Summary LSA - Path to ASBR",
    5: "External LSA - External routes"
}
```

```python
def test_ospf_neighbor_formation(device, expected_neighbors):
    """
    Test OSPF neighbor adjacency - common interview question
    """
    # Get neighbor information
    output = device.send_command("show ip ospf neighbor")
    
    # Parse output
    neighbors = parse_ospf_neighbors(output)
    
    # Validate each expected neighbor
    results = []
    for expected in expected_neighbors:
        if expected in neighbors:
            neighbor = neighbors[expected]
            if neighbor['state'] == 'FULL':
                results.append(f"✓ {expected}: FULL")
            else:
                results.append(f"✗ {expected}: {neighbor['state']}")
        else:
            results.append(f"✗ {expected}: NOT FOUND")
    
    return results
```
